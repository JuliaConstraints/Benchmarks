cmake_minimum_required(VERSION 3.15)
project(ghost_routing_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ghost_routing main.cpp)
target_include_directories(ghost_routing PRIVATE "${GHOST_SNAPSHOT}/include")
target_link_libraries(ghost_routing PRIVATE "${GHOST_SNAPSHOT}/lib/ghost_static.lib")
if(MSVC)
  target_compile_options(ghost_routing PRIVATE /O2 /EHsc)
endif()
